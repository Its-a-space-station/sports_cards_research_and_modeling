**Goal:** Run the multivariate analysis the whole project exists for — which player stats drive graded-rookie card prices, whether effects are consistent across seasons and positions, and whether a walk-forward buy signal survives an honest gate — and write up the findings.

**Architecture:** Feature-matrix builder (panel parquets → standardized modeling frame) → three estimation passes (LASSO stability selection, LightGBM + SHAP, mixed-effects + interaction tests) → walk-forward harness with a fee-aware gate → findings report. All model code is tested on synthetic data with known planted signal before touching the real panel.

**Tech Stack:** Python 3.11+, existing `cardprice` package; new deps: `scikit-learn`, `lightgbm`, `shap`, `statsmodels`, `numpy`.

**Spec:** `docs/superpowers/specs/2026-09-14-mlb-card-price-panel-design.md` (§6 is the binding section)
**Predecessor:** Plan 3 — `data/processed/panel_monthly.parquet` (325 rows, 13 cards, PSA-10, 2022-10→2026-09) and `panel_weekly.parquet` (184 rows, all grades); `docs/data_dictionary.md`.

## Facts established by Plans 1-3 (code below is built on them)

- Panel columns: keys `card_slug, mlb_id, player_name, month|week`; outcome `log_ret, months_since_prev|days_since_prev, market_median_ret, excess_ret`; hitter predictors `games, ops, avg, obp, slg, home_runs, strikeouts`; pitcher predictors `era, whip, k_bb_pct, innings_pitched`; form `form_games, form_ops_delta|form_era_delta`; static `age, position, rookie_year, stats_season, set_slug, grade, playoff`; weekly adds `n_sales, best_offer_share`.
- **Sample size reality:** ~312 monthly returns, 13 players (11 hitters, 2 pitchers: Skenes, Sasaki), 4-5 seasons. Pitchers are too few to model separately — primary analysis is hitters; pitchers are descriptive-only (disclosed in the report). The models can only detect large effects; the findings report must say so.
- **Horizon rule (Plan 3 final review):** weekly primary analysis uses `days_since_prev <= 21`; monthly flags `months_since_prev > 2` (drop from primary, keep for robustness).
- **Pop data:** one current snapshot per card (2026-09-15), no history. Using it as a predictor for 2022-2025 returns would be look-ahead (populations grow over time) — pop columns are EXCLUDED from regression features; used for descriptive stats only.
- Weekly panel covers mostly the 2026 window (recent sales) — within-season history lives in the monthly panel.

## Global Constraints

- Python >= 3.11; free sources; no paid data.
- **No look-ahead:** predictors are already lagged in the panels; modeling must not reintroduce leakage — walk-forward fits only on strictly earlier periods, never random train/test splits.
- All modeling code is unit-tested on synthetic data with planted signal BEFORE the real run; tests must show the method recovers the planted driver.
- Reproducibility: every stochastic step takes `seed=42` (or threaded seed).
- Honest reporting: the findings report states effect sizes, uncertainty, and power limits; a failed gate is a valid, reportable result.
- Deps to add to pyproject: `scikit-learn>=1.5`, `lightgbm>=4`, `shap>=0.46`, `statsmodels>=0.14`, `numpy>=2`.

```python
# src/cardprice/features.py
"""Panel parquets -> standardized modeling frames. Horizon rules per Plan 3 review."""

import numpy as np
import pandas as pd

HITTER_FEATURES = [
    "ops", "slg", "home_runs", "strikeouts", "games",
    "form_games", "form_ops_delta", "age", "playoff", "years_since_rookie",
]

POSITION_MAP = {
    "P": "P", "C": "C",
    "1B": "IF", "2B": "IF", "3B": "IF", "SS": "IF",
    "OF": "OF", "LF": "OF", "CF": "OF", "RF": "OF", "DH": "OF",
}


def build_modeling_frame(panel: pd.DataFrame, grain: str) -> pd.DataFrame:
    df = panel.copy()
    if grain == "monthly":
        df = df[(df["months_since_prev"].isna()) | (df["months_since_prev"] <= 2)]
    elif grain == "weekly":
        df = df[(df["days_since_prev"].isna()) | (df["days_since_prev"] <= 21)]
    else:
        raise ValueError(f"unknown grain: {grain}")
    df = df[df["excess_ret"].notna()]
    df["years_since_rookie"] = df["stats_season"] - df["rookie_year"]
    df["position_group"] = df["position"].map(POSITION_MAP)
    return df.reset_index(drop=True)


def standardize(X: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
    params = pd.DataFrame({"mean": X.mean(), "sd": X.std(ddof=0)})
    Z = (X - params["mean"]) / params["sd"].replace(0, 1)
    Z.loc[:, params["sd"] == 0] = 0.0
    return Z, params


def hitter_matrix(frame: pd.DataFrame) -> tuple[pd.DataFrame, pd.Series, pd.DataFrame]:
    hit = frame[frame["position_group"] != "P"]
    X = hit[HITTER_FEATURES].copy()
    X = X.fillna(X.median())
    Z, _ = standardize(X)
    y = hit["excess_ret"]
    groups = hit[["mlb_id", "position_group", "stats_season"]].reset_index(drop=True)
    return Z.reset_index(drop=True), y.reset_index(drop=True), groups
```

```python
# src/cardprice/model_lasso.py
"""LASSO with stability selection: which stats drive card price changes."""

import numpy as np
import pandas as pd
from sklearn.linear_model import LassoCV


def _fit_selected(X: pd.DataFrame, y: pd.DataFrame | pd.Series, seed: int) -> pd.Index:
    """Fit LassoCV and select with the one-standard-error rule (largest alpha whose
    CV MSE is within 1 SE of the minimum, refit via Lasso). CV-min alpha is
    prediction-optimal but selection-inconsistent — it over-selects spurious
    features (verified empirically: the planted-signal test fails with cv-min).
    (Corrected during execution; an earlier draft selected at cv-min directly.)"""
    from sklearn.linear_model import Lasso

    cv = LassoCV(cv=5, random_state=seed, max_iter=10000).fit(X, y)
    mse = cv.mse_path_.mean(axis=1)  # (n_alphas, n_folds) -> per-alpha mean
    se = cv.mse_path_.std(axis=1, ddof=1) / np.sqrt(cv.mse_path_.shape[1])
    threshold = mse.min() + se[mse.argmin()]
    within = np.flatnonzero(mse <= threshold)  # alphas_ sorted descending
    alpha_1se = cv.alphas_[within[0]]  # largest alpha within 1 SE
    coef = Lasso(alpha=alpha_1se, max_iter=10000).fit(X, y).coef_
    return X.columns[coef != 0]


def stability_selection(
    X: pd.DataFrame, y: pd.Series, n_boot: int = 200, seed: int = 42
) -> pd.Series:
    rng = np.random.default_rng(seed)
    counts = pd.Series(0, index=X.columns, dtype=float)
    n = len(X)
    for b in range(n_boot):
        idx = rng.integers(0, n, n)
        selected = _fit_selected(X.iloc[idx], y.iloc[idx], seed + b)
        counts[selected] += 1
    return (counts / n_boot).sort_values(ascending=False)


def lasso_path_summary(X: pd.DataFrame, y: pd.Series, seed: int = 42) -> pd.DataFrame:
    model = LassoCV(cv=5, random_state=seed, max_iter=10000).fit(X, y)
    out = pd.DataFrame({"feature": X.columns, "coefficient": model.coef_})
    out = out[out["coefficient"] != 0]
    return out.reindex(out["coefficient"].abs().sort_values(ascending=False).index)
```

```python
# src/cardprice/model_gbm.py
"""Gradient boosting + SHAP: nonlinear importance cross-check for the LASSO ranking."""

import lightgbm as lgb
import numpy as np
import pandas as pd
import shap
from sklearn.model_selection import KFold


def _make_model(seed: int) -> lgb.LGBMRegressor:
    return lgb.LGBMRegressor(
        n_estimators=300, learning_rate=0.05, num_leaves=15,
        subsample=0.8, random_state=seed, verbose=-1,
    )


def gbm_shap_importance(X: pd.DataFrame, y: pd.Series, seed: int = 42) -> pd.DataFrame:
    model = _make_model(seed).fit(X, y)
    sv = shap.TreeExplainer(model).shap_values(X)
    imp = pd.DataFrame(
        {"feature": X.columns, "mean_abs_shap": np.abs(sv).mean(axis=0)}
    )
    return imp.sort_values("mean_abs_shap", ascending=False).reset_index(drop=True)


def gbm_cv_r2(X: pd.DataFrame, y: pd.Series, seed: int = 42) -> float:
    kf = KFold(5, shuffle=True, random_state=seed)
    preds = pd.Series(np.nan, index=y.index)
    for train_idx, test_idx in kf.split(X):
        model = _make_model(seed).fit(X.iloc[train_idx], y.iloc[train_idx])
        preds.iloc[test_idx] = model.predict(X.iloc[test_idx])
    ss_res = ((y - preds) ** 2).sum()
    ss_tot = ((y - y.mean()) ** 2).sum()
    return 1 - ss_res / ss_tot
```

```python
# src/cardprice/model_consistency.py
"""Is a stat's effect consistent across seasons and positions?"""

import numpy as np
import pandas as pd
import statsmodels.formula.api as smf


def season_consistency(frame: pd.DataFrame, stat: str) -> pd.DataFrame:
    rows = []
    for season, grp in frame.groupby("stats_season"):
        grp = grp.dropna(subset=[stat, "games", "age", "excess_ret"])
        if len(grp) < 5:
            rows.append({"season": int(season), "coef": np.nan, "se": np.nan,
                         "p_value": np.nan, "n": len(grp)})
            continue
        fit = smf.ols(
            f"excess_ret ~ {stat} + games + age", data=grp
        ).fit(cov_type="HC1")
        rows.append(
            {
                "season": int(season),
                "coef": fit.params[stat],
                "se": fit.bse[stat],
                "p_value": fit.pvalues[stat],
                "n": len(grp),
            }
        )
    return pd.DataFrame(rows)


def position_interaction_test(frame: pd.DataFrame, stat: str) -> dict:
    df = frame[frame["position_group"].isin(["IF", "OF"])].dropna(
        subset=[stat, "games", "age", "excess_ret"]
    )
    fit = smf.ols(
        f"excess_ret ~ {stat} * C(position_group) + games + age", data=df
    ).fit(cov_type="HC1")
    interaction_terms = [t for t in fit.params.index if ":" in t]
    if not interaction_terms:
        return {"f_stat": np.nan, "p_value": np.nan}
    restriction = " and ".join(f"{t} = 0" for t in interaction_terms)
    ftest = fit.f_test(restriction)  # joint F-test: all interaction terms are zero
    return {"f_stat": float(ftest.fvalue), "p_value": float(ftest.pvalue)}


def player_random_effects(frame: pd.DataFrame, stat: str) -> dict:
    df = frame.dropna(subset=[stat, "games", "age", "excess_ret"])
    fit = smf.mixedlm(
        f"excess_ret ~ {stat} + games + age", data=df, groups=df["mlb_id"]
    ).fit(reml=True)
    ri_var = float(fit.cov_re.iloc[0, 0])
    resid_var = float(fit.scale)
    return {
        "random_intercept_var": ri_var,
        "resid_var": resid_var,
        "icc": ri_var / (ri_var + resid_var) if (ri_var + resid_var) > 0 else 0.0,
    }
```

If `f_test` with the "A = 0 and B = 0" restriction string fails, the equivalent is a restriction matrix; make the two planted-signal tests pass as written and document the approach in your report. All three model functions drop NaN rows explicitly (form columns are legitimately NaN early in a season).

```python
# src/cardprice/walkforward.py
"""Walk-forward evaluation: fit on months < m, predict month m, gate on realized excess."""

import numpy as np
import pandas as pd
from sklearn.linear_model import LassoCV


def walk_forward(
    frame: pd.DataFrame, feature_cols: list[str], top_k: int = 2, min_train: int = 6, seed: int = 42
) -> pd.DataFrame:
    months = sorted(frame["month"].unique())
    rows = []
    for mi, m in enumerate(months):
        if mi < min_train:
            continue
        train = frame[frame["month"] < m]
        test = frame[frame["month"] == m]
        medians = train[feature_cols].median()
        X_train = train[feature_cols].fillna(medians)
        X_test = test[feature_cols].fillna(medians)
        model = LassoCV(cv=5, random_state=seed, max_iter=10000).fit(X_train, train["excess_ret"])
        preds = model.predict(X_test)
        ranked = test.assign(predicted=preds).nlargest(top_k, "predicted")
        for r in ranked.itertuples():
            rows.append(
                {"month": m, "card_slug": r.card_slug, "predicted": r.predicted,
                 "realized": r.excess_ret}
            )
    return pd.DataFrame(rows)


def gate_evaluation(
    picks: pd.DataFrame, frame: pd.DataFrame, fee_log: float = 0.14,
    n_boot: int = 1000, seed: int = 42,
) -> dict:
    monthly = []
    for m, grp in picks.groupby("month"):
        strategy = grp["realized"].mean()
        benchmark = frame[frame["month"] == m]["excess_ret"].median()
        monthly.append(strategy - benchmark)
    monthly = np.array(monthly)
    rng = np.random.default_rng(seed)
    boot = rng.choice(monthly, size=(n_boot, len(monthly)), replace=True).mean(axis=1)
    return {
        "mean_excess": float(monthly.mean()),
        "ci_low": float(np.percentile(boot, 2.5)),
        "ci_high": float(np.percentile(boot, 97.5)),
        "p_value": float((boot <= 0).mean()),
        "net_of_fees_mean": float(monthly.mean() - fee_log),
        "n_months": len(monthly),
    }
```
